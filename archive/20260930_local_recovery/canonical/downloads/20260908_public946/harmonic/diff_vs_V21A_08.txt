+def weakly_connected_components(node_ids, edges):
+    parent = {n: n for n in node_ids}
+
+    def find(x):
+        while parent[x] != x:
+            parent[x] = parent[parent[x]]
+            x = parent[x]
+        return x

+    def union(a, b):
+        ra, rb = find(a), find(b)

+            parent[ra] = rb
